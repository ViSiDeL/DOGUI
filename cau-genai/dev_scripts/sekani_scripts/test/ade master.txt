;
;  STUD FRAME WALL PROGRAM
;
;
;  Function ade:
;     Begin actual drawing of structure. (Part 1 of 6)
;
; Date: Nov. 17 1984
;
(defun C:ade1 ()
;
;  Set sole plate (sp) variables.
;
   (setq sp-length (getstring "Set sole plate length (IN DECIMAL FEET): "))
;
   (setq stud-space
       (getreal "Set center-to-center distance between studs (1.5' or 2') : "))
   (while (and (/= stud-space 1.5) (/= stud-space 2))
       (progn
       (prompt "\n")
       (prompt "No, dummy, wrong value. Enter number with no units.")
       (prompt "\n")
       (setq stud-space
       (getreal "Set center-to-center distance between studs (1.5' or 2') : "))
       )
   )
;
   (while (> (atof sp-length) 90)
       (progn
       (prompt "\n")
       (prompt "No, dummy, that's too big. Enter number with no units.")
       (prompt "\n")
       (setq sp-length (getstring "Set sole plate length (IN DECIMAL FEET): "))
       )
   )
   (setq number-of-studs (/ (- (atof sp-length) 0.1667) stud-space))
   (setq save-number-of-studs number-of-studs)
   (setq number-of-studs (+ number-of-studs 1))
;
   (setq number-of-studs (atoi (rtos number-of-studs 2 2)))
;
   (setq sp-thickness 0.1667)
   (setq sp-elev 0)
   (setq sp-width "0.25")
;
   (setq set-limits (rtos (+ (atof sp-length) 20) 2 4))
   (setq upper-limits (strcat set-limits "," set-limits))
;
;
;
)

;
;  STUD FRAME WALL PROGRAM
;  GENERIC
;
;
;  Function ade2:
;     Begin actual drawing of structure. (Part 2 of 6)
;
;  Date: Nov. 20 1984
;
;
(defun C:ade2 ()
;
;  Set up drawing parameters.
;
   (command "LIMITS" "0,0" upper-limits)
   (if (<= (atof set-limits) 30) (setq set-grid "1"))
   (if (and (<= (atof set-limits) 50) (> (atof set-limits) 30))
       (setq set-grid "2"))
   (if (and (<= (atof set-limits) 100) (> (atof set-limits) 50))
       (setq set-grid "5"))
   (command "GRID" set-grid)
;
;  Set up start point according to sole plate size.
;
   (setq start-pt-y (- (/ (atof set-limits) 2) 0.125))
   (setq start-pt-x 5)
   (setq start-pt (list start-pt-x start-pt-y))
;
;  Begin drawing sole plate.
;
   (command "ELEV" sp-elev sp-thickness)
   (command "PLINE" start-pt (strcat "@" sp-length "<0")
			     (strcat "@" sp-width "<90")
			     (strcat "@" sp-length "<180")
			     "close")
;
;  Set up start point for leader.
;
   (setq star-pt-y (- (/ (atof set-limits) 2) 0.0625))
   (setq star-pt-x (+ start-pt-x (atof sp-length)))
   (setq star-pt (list star-pt-x star-pt-y))
;
   (command "ELEV" "0" "0")
   (command "DIM" "LEADER" star-pt (strcat "@" "2.345" "<0")
				   ""
				   "2 x 4 WOOD STUD SYSTEM"
				   "EXIT")
;
;
;  Set up view point.
;
   (command "GRID" "OFF")
   (command "VPOINT" "2,-2,2")
   (command "ZOOM" "ALL")
)

ADE 3:
;
;  STUD FRAME WALL PROGRAM
;  GENERIC
;
;
;  Function ade3:
;     Begin actual drawing of structure. (Part 3 of 6)
;
;  Date: Nov. 30 1989
;
;
(defun C:ade3 ()
;
;  Set stud variables.
;
   (setq stud-thickness (getint "Set stud height (8' or 10') : "))
   (while (and (/= stud-thickness 8) (/= stud-thickness 10))
      (progn
      (prompt "\n")
      (prompt "Wrong value, dummy. Enter number with no unit.")
      (prompt "\n")
      (setq stud-thickness (getint "Set stud height (8' or 10') : "))
      )
   )
   (setq stud-elev 0.1667)
   (setq stud-width "0.25")
   (setq stud-length "0.1667")
;
;  Begin drawing studs.
;
   (command "ELEV" stud-elev stud-thickness)
   (setq counter (1- number-of-studs))
;
;  Save value of initial start point, START-PT.
;
   (setq save-start-pt start-pt)
;
   (while (> counter 0)
      (command "PLINE" start-pt (strcat "@" stud-length "<0")
				(strcat "@" stud-width "<90")
				(strcat "@" stud-length "<180")
				"close")
;
;  Set X-START-PT to the x-value of START-PT plus STUD-SPACE.
;  (car START-PT) gets the x-value of START-PT (x,y).
;  (cadr START-PT) gets the y-value of START-PT (x,y).
;  x-value increases and y-value stays the same.
;  (LIST X-START-PT Y-START-PT) creates these values into an (x,y) pair.
;
      (setq x-start-pt (+ (car start-pt) stud-space))
      (setq y-start-pt (cadr start-pt))
      (setq start-pt (list x-start-pt y-start-pt))
      (setq counter (1- counter))
   )
;
;  Draw the last stud beginning at the lower right point of the sole plate.
;
   (setq lower-right-x (+ 5 (atof sp-length)))
   (setq lower-right-y start-pt-y)
   (setq lower-right-pt (list lower-right-x lower-right-y))
;
   (command "PLINE" lower-right-pt (strcat "@" stud-width "<90")
				   (strcat "@" stud-length "<180")
				   (strcat "@" stud-width "<-90")
				   "close")
;
;  Make view easier to look at.
;
   (command "ELEV" "0" "0")
   (command "ZOOM" "All")
;
)

;     Begin calculating dimensions and prices.
;
;  Written:  March 1988

(defun C:CALC ()
;
;  Calculate dimensions of materials.
;  Calculate number of board feet of lumber.
;
   (setq length-of-lumber (+ (atof sp-length) (* save-number-of-studs
			  (atof stud-length)) (* 2 (atof hd-length))))

   (setq volume-of-lumber (* length-of-lumber sp-thickness (atof sp-width)))
   (setq volume-in-inches (* volume-of-lumber 1728))
   (setq number-of-bdft (/ volume-in-inches 144))
;
;  Calculate square feet of sheet rock.
;
   (setq sqft-sheet-rock (* gp-thickness (atof gp-length)))
;
;  Calculate square feet of insulation.
;
   (setq sqft-insulation (* ins-thickness (atof ins-length)))
;
;  Calculate square feet of brick and mortar.
;
   (setq sqft-brick-mortar (* brick-thickness (atof brick-length)))
;
;
;
;  Draw border lines around wall statistics area.
;
;
   (command "VPOINT" "0,0,1")
   (command "ZOOM" "C" "4,5.5" "11")
   (command "PLINE" "0,0" (strcat "@" "10" "<0")
			  (strcat "@" "11.0" "<90")
			  (strcat "@" "10" "<180")
			  "close")
;
;
   (command "PLINE" "0.2,0.2" (strcat "@" "9.6" "<0")
			      (strcat "@" "10.6" "<90")
			      (strcat "@" "9.6" "<180")
			      "close")


;  Print dimensions to screen.
;
   (setq p1 (rtos number-of-bdft 2 2))
   (setq p2 (rtos sqft-sheet-rock 2 2))
   (setq p3 (rtos sqft-insulation 2 2))
   (setq p4 (rtos sqft-brick-mortar 2 2))

def ade1():
    # Set sole plate (sp) variables.
    sp_length = input("Set sole plate length (IN DECIMAL FEET): ")

    stud_space = float(input("Set center-to-center distance between studs (1.5' or 2') : "))
    while stud_space != 1.5 and stud_space != 2:
        print("\nNo, dummy, wrong value. Enter number with no units.")
        stud_space = float(input("Set center-to-center distance between studs (1.5' or 2') : "))

    while float(sp_length) > 90:
        print("\nNo, dummy, that's too big. Enter number with no units.")
        sp_length = input("Set sole plate length (IN DECIMAL FEET): ")

    number_of_studs = (float(sp_length) - 0.1667) / stud_space
    save_number_of_studs = number_of_studs
    number_of_studs += 1

    number_of_studs = round(number_of_studs)

    sp_thickness = 0.1667
    sp_elev = 0
    sp_width = "0.25"

    set_limits = f"{round(float(sp_length) + 20, 2):.4f}"
    upper_limits = f"{set_limits},{set_limits}"
